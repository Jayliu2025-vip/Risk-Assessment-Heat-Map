```python
# -*- coding: utf-8 -*-
import json
import re
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
DIMS = {
    "imp_financial", "imp_compliance", "imp_operation", "imp_reputation",
    "imp_fraud", "imp_strategy", "imp_data", "imp_hse",
}


def read_text(relative):
    return (ROOT / relative).read_text(encoding="utf-8")


def embedded_sample():
    text = read_text("web/sample_data.js").strip()
    prefix = "window.SAMPLE_DATA = "
    if not text.startswith(prefix) or not text.endswith(";"):
        raise AssertionError("web/sample_data.js is not a SAMPLE_DATA assignment")
    return json.loads(text[len(prefix):-1])


class ReleaseConsistencyTests(unittest.TestCase):
    def test_current_user_copy_has_no_legacy_model_counts(self):
        forbidden = {
            "README.md": ("21 项",),
            "docs/使用手册.md": ("五维", "六项打分", "16 条", "6 个领域", "21 项"),
            "tools/build_excel.py": ("五维", "六项打分"),
            "tools/generate_report.py": ("五维加权",),
            "web/risk_heatmap.html": ("五维影响权重", "五维加权"),
        }
        for relative, phrases in forbidden.items():
            text = read_text(relative)
            for phrase in phrases:
                with self.subTest(file=relative, phrase=phrase):
                    self.assertNotIn(phrase, text)

    def test_web_default_weights_cover_all_dimensions(self):
        html = read_text("web/risk_heatmap.html")
        match = re.search(
            r"const DEFAULT_CFG\s*=\s*\{\s*.*?weights:\{([^}]*)\}",
            html,
            re.S,
        )
        self.assertIsNotNone(match)
        pairs = re.findall(r"(imp_[a-z]+):([0-9.]+)", match.group(1))
        weights = {key: float(value) for key, value in pairs}
        self.assertEqual(set(weights), DIMS)
        self.assertAlmostEqual(sum(weights.values()), 1.0, places=6)

    def test_web_uses_exported_floor_key(self):
        html = read_text("web/risk_heatmap.html")
        self.assertIn("impact_floor_factor:.75", html)
        self.assertIn("cfg.impact_floor_factor", html)

    def test_embedded_sample_config_matches_export_config(self):
        exported = json.loads(read_text("data/export/config.json"))
        self.assertEqual(embedded_sample()["config"], exported)


if __name__ == "__main__":
    unittest.main()
```

```javascript
const DEFAULT_CFG = {
  version:"1.1",
  weights:{imp_financial:.25,imp_compliance:.20,imp_operation:.12,imp_reputation:.10,
    imp_fraud:.10,imp_strategy:.10,imp_data:.08,imp_hse:.05},

  reduction_map:{"1":0,"2":.15,"3":.4,"4":.55,"5":.7},
  thresholds:{extreme:20,high:12,medium:6,low:3}
};

function normalizeConfig(raw={}){
  const cfg=JSON.parse(JSON.stringify(DEFAULT_CFG));
  cfg.version=raw.version??cfg.version;
  cfg.weights={...cfg.weights,...(raw.weights||{})};
  cfg.domain_weights={...(raw.domain_weights||{})};
  cfg.reduction_map={...cfg.reduction_map,...(raw.reduction_map||{})};
  cfg.thresholds={...cfg.thresholds,...(raw.thresholds||{})};
  cfg.impact_floor_factor=+(raw.impact_floor_factor??raw.impact_floor??cfg.impact_floor_factor);
  cfg.ref_reduction=+(raw.ref_reduction??.4);
  return cfg;
}
```

```powershell
@'
import csv, json
from pathlib import Path
from openpyxl import load_workbook
from tools.export_from_excel import read_config, read_rows

root = Path('.')

    risks.append({'risk_id': str(vals[0]).strip(), 'name': vals[1], 'domain': vals[2],
                  'description': vals[3] or '', 'owner_dept': vals[4] or '', 'period': vals[5],

controls = []
for vals in read_rows(wb['控制措施表'], 6):
    controls.append({'control_id': str(vals[0]).strip(), 'risk_id': str(vals[1]).strip(),
                     'period': vals[2], 'description': vals[3] or '', 'score': int(vals[4]),
                     'key': '是' if str(vals[5]).strip() in ('是','1','true','True') else '否'})

for period in sorted({r['period'] for r in risks}):
    with (root / f'data/export/{period}/risks.csv').open(encoding='utf-8-sig', newline='') as f:
```
